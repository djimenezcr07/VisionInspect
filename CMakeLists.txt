cmake_minimum_required(VERSION 3.20)
project(VisionInspect VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
if(OpenCV_VERSION VERSION_GREATER_EQUAL "5.0")
  find_package(OpenCV REQUIRED COMPONENTS core imgproc geometry)
endif()
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module NumPy)
find_package(pybind11 CONFIG REQUIRED)
add_library(vision_core STATIC cpp/src/inspection.cpp)
target_include_directories(vision_core PUBLIC cpp/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_core PUBLIC ${OpenCV_LIBS})
set_target_properties(vision_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
pybind11_add_module(vision_engine cpp/src/bindings.cpp)
target_link_libraries(vision_engine PRIVATE vision_core)
enable_testing()
add_executable(vision_tests cpp/tests/test_engine.cpp)
target_link_libraries(vision_tests PRIVATE vision_core)
add_test(NAME vision_core COMMAND vision_tests)
